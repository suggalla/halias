/**
 * @title HaliasPaymaster
 * @dev Sponsors two classes of UserOp:
 *
 *   PATH_POOL_NOTE (0x00): transact() with recipient=paymaster (pool-funded gas).
 *     Gas budget = absAmount from pool note. postOp deposits absAmount back to
 *     EntryPoint; overpay (absAmount - actualGasCost) builds a revolving reserve.
 *
 *   PATH_VOUCHER (0x01): registerWithVoucher() for zero-ETH new users.
 *     Validates the voucher secret and that the UserOp sender is the designated
 *     beneficiary. postOp is a no-op — EntryPoint handles the deposit accounting.
 *
 * paymasterAndData layout:
 *   [0:20]  paymaster address
 *   [20]    path type (0x00 = pool-note, 0x01 = voucher)
 *   [21:86] 65-byte relayer sig (pool-note path, when relayer != address(0))
 *   [21:53] 32-byte voucher secret (voucher path)
 */

    function postOp(
        IPaymaster.PostOpMode mode,
        bytes calldata context,
        uint256,
        uint256
    ) external override {
        if (msg.sender != address(entryPoint)) revert NotEntryPoint();
 
        (uint8 pathType, bytes32 data) = abi.decode(context, (uint8, bytes32));
 
        if (pathType == PATH_POOL_NOTE) {
            // Pool-note path: deposit absAmount from the transact() ETH received.
            // The surplus (absAmount - actualGasCost) builds the revolving reserve.
            if (mode == IPaymaster.PostOpMode.opSucceeded) {
                uint256 absAmount = uint256(data);
                if (absAmount > 0) entryPoint.depositTo{value: absAmount}(address(this));
            }
            // opReverted: transact() did not execute; pool note not spent; do nothing.
        } else {

            // this paymaster's EP during execution. The surplus (gasBudget - actualGasCost)
            // stays here as accumulated protocol fee — the admin sweeps it via
            // withdrawEntryPointDeposit. Nothing to route; the EP balance is the ledger.
        }
    }
