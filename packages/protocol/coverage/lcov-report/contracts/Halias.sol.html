        uint256 publicAmount,
        uint256 tokenAddress,
        TransactParams calldata params
    ) internal {

        uint256 absAmount = isWithdraw ? (FIELD_PRIME - publicAmount) : publicAmount;
 
        if (tokenAddress == 0) {
            if (isWithdraw) {
                if (msg.value != 0) revert WithdrawCannotHaveValue();
                if (params.recipient == address(0)) revert NoDestination();

                Eif (params.recipient != address(this)) {
                    payable(params.recipient).sendValue(absAmount);
                    emit Withdrawal(params.recipient, absAmount, 0);
                }
            } else {
                if (publicAmount > 0) {
                    if (msg.value != publicAmount) revert WrongDepositValue();
                } else {
                    if (msg.value != 0) revert TransferCannotHaveValue();
                }
            }
        } else {
            if (msg.value != 0) revert ERC20CannotHaveETH();
            address token = address(uint160(tokenAddress));
            if (token.code.length == 0) revert InvalidTokenAddress();

                if (params.recipient == address(0)) revert NoDestination();
                poolTokenBalance[token] -= absAmount;
                IERC20(token).safeTransfer(params.recipient, absAmount);
                emit Withdrawal(params.recipient, absAmount, tokenAddress);

                IERC20(token).safeTransferFrom(msg.sender, address(this), publicAmount);
                if (IERC20(token).balanceOf(address(this)) < balBefore + publicAmount) revert FeeOnTransferToken();

        }

    // Recover ERC-20 tokens sent directly to this contract via token.transfer() rather

    // poolTokenBalance tracks the pool's exact ERC-20 liability (sum of all outstanding
