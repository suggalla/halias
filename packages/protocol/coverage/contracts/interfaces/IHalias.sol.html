// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;
 
// External calldata type for transact() and transactWithPermit().
// Bound to the ZK proof via paramsHash = keccak256(chainId, contract, fields...) % FIELD_PRIME.
struct TransactParams {
    address recipient;       // unshield destination; address(this) = retain ETH for paymaster gas
    bytes encryptedOutput0;
    bytes encryptedOutput1;
    bytes32 externalData;    // general-purpose commitment hook (compliance, relayer fees, etc.)
}
 
interface IHalias {
    // ── Registration ───────────────────────────────────────────────────────────
    function register(bytes32 aliasHash, bytes32 spendingPubkey, bytes32 nullifierKey, bytes32 encryptionPubkey) external payable;
    function registerWithVoucher(bytes32 aliasHash, bytes32 spendingPubkey, bytes32 nullifierKey, bytes32 encryptionPubkey, bytes32 secret) external;
    function updateKeys(bytes32 aliasHash, bytes32 newNullifierKey, bytes32 newEncryptionPubkey) external;
    function updateAliasData(bytes32 aliasHash, bytes32 newDataHash) external;
    function addAttestation(bytes32 aliasHash, address provider, bytes calldata proof) external;
    function updateAttestation(bytes32 aliasHash, address provider, bytes calldata proof) external;
    function removeAttestation(bytes32 aliasHash, address provider) external;
    function getProviders(bytes32 aliasHash) external view returns (address[] memory);
    function attestations(bytes32 aliasHash, address provider) external view returns (bytes32 commitment);
    function transferAliasWithKeys(bytes32 aliasHash, address newOwner, bytes32 newSpendingPubkey, bytes32 newNullifierKey, bytes32 newEncryptionPubkey) external;
 
    // ── Transact ───────────────────────────────────────────────────────────────
    function transact(bytes32 poolRoot, bytes32 registryRoot, uint256 publicAmount, uint256 tokenAddress, bytes32[2] calldata inputNullifier, bytes32[2] calldata outputCommitment, TransactParams calldata params, bytes calldata proof) external payable;
    function transactWithPermit(bytes32 poolRoot, bytes32 registryRoot, uint256 publicAmount, uint256 tokenAddress, bytes32[2] calldata inputNullifier, bytes32[2] calldata outputCommitment, TransactParams calldata params, bytes calldata proof, uint256 deadline, uint8 v, bytes32 r, bytes32 s) external payable;
    function computeParamsHash(TransactParams calldata params) external view returns (uint256);
 
    // ── Vouchers ───────────────────────────────────────────────────────────────
    function createVoucher(bytes32 commitment, uint256 expiry) external payable;
    function reclaimVoucher(bytes32 commitment) external;
    function vouchers(bytes32 commitment) external view returns (uint256 amount, uint256 expiry, address creator);
 
    // ── State queries ──────────────────────────────────────────────────────────
    function registrationFee() external view returns (uint256);
    function voucherFee() external view returns (uint256);
    function spentNullifiers(bytes32 nullifier) external view returns (bool);
    function poolTokenBalance(address token) external view returns (uint256);
    function smtRoot() external view returns (bytes32);
    function getRegistryRoot() external view returns (bytes32);
    function getSmtSiblings(uint256 key) external view returns (bytes32[32] memory);
    function isKnownPoolRoot(bytes32 root) external view returns (bool);
    function isKnownRegistryRoot(bytes32 root) external view returns (bool);
    function getLastRoot() external view returns (bytes32);
}
